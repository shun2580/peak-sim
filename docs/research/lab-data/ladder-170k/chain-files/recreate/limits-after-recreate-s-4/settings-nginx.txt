# configuration file /etc/nginx/nginx.conf:
# nginx of the lab: in front of php-fpm, on the internal network only. Every request goes to the one entry script of
# the lab (poc/lab/app/entry.php); the app's own public/index.php is never run.
worker_processes 2;
pid /tmp/nginx.pid;
error_log /dev/stderr warn;

events {
    worker_connections 4096;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;
    sendfile on;
    keepalive_timeout 65;

    client_body_temp_path /tmp/nginx-client;
    proxy_temp_path /tmp/nginx-proxy;
    fastcgi_temp_path /tmp/nginx-fastcgi;
    uwsgi_temp_path /tmp/nginx-uwsgi;
    scgi_temp_path /tmp/nginx-scgi;

    # time (seconds since the epoch, milliseconds), status, request seconds, upstream seconds, method, uri
    log_format lab '$msec $status $request_time $upstream_response_time $request_method $request_uri';

    upstream lab_app {
        server lab-app:9000;
        keepalive 64;
    }

    server {
        listen 8080;
        access_log /out/web/access.log lab;

        location / {
            include /etc/nginx/fastcgi_params;
            fastcgi_pass lab_app;
            fastcgi_keep_conn on;
            fastcgi_param SCRIPT_FILENAME /peak-sim/poc/lab/app/entry.php;
            fastcgi_param SCRIPT_NAME /index.php;
            fastcgi_read_timeout 1800s;
            fastcgi_buffering on;
            fastcgi_buffers 64 16k;
        }
    }

    # Not the app: what the lab asks the web server itself (a counter of the requests it took, a health check).
    server {
        listen 8081;
        access_log off;

        location = /__lab/status {
            stub_status;
        }

        location = /__lab/health {
            default_type text/plain;
            return 200 "ok\n";
        }
    }
}
